## Deploy Locally

First, install dependecies

```
npm i
```

Create .env file
```
NEXT_PUBLIC_SUPABASE_URL=YOUR_URL
NEXT_PUBLIC_SUPABASE_ANON_KEY=YOUR_KEY

NEXTAUTH_URL="http://localhost:3000/"
NEXTAUTH_SECRET="YOUR_SECRET"

AUTH_GITHUB_ID="YOUR_ID"
AUTH_GITHUB_SECRET="YOUR_SECRET"

NEXT_PUBLIC_GOOGLE_MAPS=YOUR_KEY

NEXT_PUBLIC_STRIPE_PUBLIC_KEY=pk_test_
STRIPE_SECRET_KEY=sk_test_
```
## Database 
```
create table public.rides (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  "startingPointAddress" text null,
  departure timestamp without time zone null,
  description text null,
  "ishaYogaCenter" text null,
  luggage text null,
  "pricePerSeat" bigint null,
  "rideDistanceMeters" bigint null,
  seats text null,
  "startingCity" text null,
  "createdByUser" uuid null,
  "startingPointCoordinates" json null,
  "rideDuration" jsonb null,
  "seatsRemaining" bigint null,
  constraint rides_pkey primary key (id)
) TABLESPACE pg_default;

create table public.bookings (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  ride_id bigint null,
  "userId" uuid null,
  seats_booked bigint null,
  "totalPrice" double precision null,
  payment_intent text not null,
  constraint bookings_pkey primary key (payment_intent),
  constraint bookings_payment_intent_key unique (payment_intent),
  constraint bookings_ride_id_fkey foreign KEY (ride_id) references rides (id)
) TABLESPACE pg_default;

```

npm run dev


Open [http://localhost:3000](http://localhost:3000) with your browser to see the result.
